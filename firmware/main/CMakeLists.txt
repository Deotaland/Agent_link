include("${CMAKE_CURRENT_LIST_DIR}/../cmake/profile_build.cmake")
set(APP_DIR "${FIRMWARE_ROOT}/apps/${PROFILE_app}")
file(GLOB _app_sources "${APP_DIR}/*.cc" "${APP_DIR}/*.cpp" "${APP_DIR}/*.c")
set(_includes "." "${FIRMWARE_ROOT}/apps/common" "${APP_DIR}" "${CMAKE_BINARY_DIR}/generated"
              "${FIRMWARE_ROOT}/boards/${PROFILE_board}")
set(_requires board_support)
if(PROFILE_backend STREQUAL "NONE")
    list(APPEND _requires esp_app_format)
endif()
if(NOT PROFILE_backend STREQUAL "NONE")
    list(APPEND _requires agent_link nvs_flash esp_adc)
    list(APPEND _app_sources app_main.cpp)
endif()
if(PROFILE_ui STREQUAL "lvgl")
    list(APPEND _requires ui_lvgl)
endif()
if(PROFILE_backend STREQUAL "MUSE")
    list(APPEND _requires esp_wifi)
endif()
set(_avatar "${FIRMWARE_ROOT}/boards/${PROFILE_board}/avatar/muse_pixel.c")
if(EXISTS "${_avatar}")
    list(APPEND _app_sources "${_avatar}")
    list(APPEND _includes "${FIRMWARE_ROOT}/boards/${PROFILE_board}")
endif()
idf_component_register(SRCS ${_app_sources} INCLUDE_DIRS ${_includes}
                       REQUIRES ${_requires})
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-missing-field-initializers)
if(EXISTS "${_avatar}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE VOICE_UI_AVATAR=1)
endif()
