    set(APP_DIR "${FIRMWARE_ROOT}/${IMAGE_path}")
    set(PROFILE_app_shared "${IMAGE_shared_app}")
endif()

set(_includes "." "${FIRMWARE_ROOT}/apps/common" "${APP_DIR}" "${CMAKE_BINARY_DIR}/generated"
              "${FIRMWARE_ROOT}/boards/${PROFILE_board}")
if(NOT PROFILE_factory STREQUAL "")
    image_list(_extra_sources sources)
    foreach(_source IN LISTS _extra_sources)
        list(APPEND _app_sources "${APP_DIR}/${_source}")
    endforeach()
    image_list(_extra_includes include_dirs)
    foreach(_include IN LISTS _extra_includes)
        list(APPEND _includes "${APP_DIR}/${_include}")
    endforeach()
endif()
if(NOT PROFILE_app_shared STREQUAL "")
    set(_shared_app "${FIRMWARE_ROOT}/apps/common/${PROFILE_app_shared}")
    file(GLOB _shared_sources "${_shared_app}/*.cc" "${_shared_app}/*.cpp" "${_shared_app}/*.c")
    list(APPEND _app_sources ${_shared_sources})
    list(APPEND _includes "${_shared_app}")
endif()
set(_requires board_support)
if(NOT PROFILE_factory STREQUAL "")
    list(APPEND _requires agent_system)
endif()
if(NOT PROFILE_ui_component STREQUAL "")
    list(APPEND _requires "${PROFILE_ui_component}")
endif()
profile_list(_app_requires app_requires)
foreach(_dependency IN LISTS _app_requires)
    if(_dependency STREQUAL "json" AND IDF_VERSION_MAJOR GREATER_EQUAL 6)
        list(APPEND _requires cjson)
    else()
        list(APPEND _requires "${_dependency}")
    endif()
endforeach()
if(PROFILE_backend STREQUAL "NONE")
    list(APPEND _requires esp_app_format)
endif()
if(NOT PROFILE_backend STREQUAL "NONE")
    list(APPEND _requires agent_link nvs_flash esp_adc)
    list(APPEND _app_sources app_main.cpp)
endif()
if(PROFILE_ui STREQUAL "lvgl")
    list(APPEND _requires ui_lvgl)
endif()
if(PROFILE_backend STREQUAL "MUSE")
    list(APPEND _requires esp_wifi)
endif()

    list(APPEND _app_sources "${_avatar}")
    list(APPEND _includes "${FIRMWARE_ROOT}/boards/${PROFILE_board}")
endif()

                       REQUIRES ${_requires})

if(EXISTS "${_avatar}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE VOICE_UI_AVATAR=1)
endif()
