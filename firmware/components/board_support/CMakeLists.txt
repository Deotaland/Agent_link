include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/profile_build.cmake")
profile_list(_drivers drivers)
profile_list(_hardware_sources hardware_sources)
set(_sources "")
foreach(_file IN LISTS _drivers)
    list(APPEND _sources "${FIRMWARE_ROOT}/boards/common/${_file}")
endforeach()

    list(APPEND _sources "${FIRMWARE_ROOT}/boards/${PROFILE_hardware}/${_file}")
endforeach()

    INCLUDE_DIRS "${FIRMWARE_ROOT}/boards/common" "${FIRMWARE_ROOT}/boards/${PROFILE_hardware}"
