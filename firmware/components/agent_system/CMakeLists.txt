include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/profile_build.cmake")
if(PROFILE_factory STREQUAL "")
    idf_component_register()
    return()
endif()
if(IDF_VERSION_MAJOR GREATER_EQUAL 6)
    set(_json cjson)
else()
    set(_json json)
endif()
set(_sources "agent_system.cc" "cover_cache.cc")
set(_store_requires "")
string(JSON _store_origin ERROR_VARIABLE _store_error GET "${BUILD_PROFILE}" community_store origin)
if(NOT _store_error)
    string(JSON _store_product GET "${BUILD_PROFILE}" community_store product_id)
    string(JSON _store_revision GET "${BUILD_PROFILE}" community_store hardware_revision_id)
    list(APPEND _sources "community_store.cc" "community_cover.cc")
    list(APPEND _store_requires esp_jpeg webp_decoder png_decoder)
endif()
string(JSON _dynamic ERROR_VARIABLE _dynamic_error GET "${BUILD_PROFILE}" layout_info dynamic_slots max_slots)
string(JSON _launcher_ab ERROR_VARIABLE _ab_error GET "${BUILD_PROFILE}" layout_info launcher_ab)
if(NOT _ab_error)
    list(APPEND _sources "launcher_update.cc")
endif()
if(NOT _dynamic_error)
    list(APPEND _sources "dynamic_slots.cc")
    execute_process(COMMAND "${PYTHON}" "${FIRMWARE_ROOT}/tools/dynamic_layout.py"
        --profile "$ENV{AGENT_LINK_BUILD_PROFILE}" --header "${CMAKE_CURRENT_BINARY_DIR}/dynamic_layout_config.h"
        RESULT_VARIABLE _result ERROR_VARIABLE _error)
    if(NOT _result EQUAL 0)
        message(FATAL_ERROR "${_error}")
    endif()
    string(JSON _dynamic_csv GET "${BUILD_PROFILE}" layout_info csv)
    get_filename_component(_dynamic_csv_path "${FIRMWARE_ROOT}/${_dynamic_csv}" ABSOLUTE)
    get_filename_component(_dynamic_tool_path "${FIRMWARE_ROOT}/tools/dynamic_layout.py" ABSOLUTE)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${_dynamic_csv_path}" "${_dynamic_tool_path}")
endif()
idf_component_register(SRCS ${_sources} INCLUDE_DIRS "include"
    PRIV_INCLUDE_DIRS "../agent_link/src/ota"
    REQUIRES esp_common esp_partition esp_app_format
    PRIV_REQUIRES app_update bootloader_support nvs_flash esp_http_client esp_http_server
                  esp_netif esp_timer heap mbedtls spi_flash esp_rom ${_json} ${_store_requires})
if(NOT _store_error)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC AGENT_SYSTEM_COMMUNITY=1)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        AGENT_COMMUNITY_ORIGIN="${_store_origin}" AGENT_COMMUNITY_PRODUCT=${_store_product}
        AGENT_COMMUNITY_REVISION=${_store_revision})
endif()
string(JSON _model GET "${SYSTEM_MANIFEST}" model)
string(JSON _layout GET "${SYSTEM_MANIFEST}" layout)
string(JSON _slots LENGTH "${BUILD_PROFILE}" layout_info app_slots)
if(NOT _dynamic_error)
    set(_slots ${_dynamic})
    string(JSON _user_slots ERROR_VARIABLE _user_error GET "${BUILD_PROFILE}" layout_info dynamic_slots user_slots)
    if(NOT _user_error)
        set(_slots ${_user_slots})
    endif()
    target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE AGENT_SYSTEM_DYNAMIC_SLOTS=1)
endif()
if(NOT _ab_error)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC AGENT_LAUNCHER_AB=1)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${FIRMWARE_ROOT}/factory/sdgoods-esp32s3")
    if(AGENT_LINK_IMAGE_ROLE STREQUAL "factory")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE AGENT_LAUNCHER_IMAGE=1)
    endif()
endif()
string(JSON _reserve GET "${BUILD_PROFILE}" layout_info cover_reserve)
string(JSON _pool GET "${BUILD_PROFILE}" layout_info app_slots 0)
string(JSON _slot_size GET "${BUILD_PROFILE}" layout_info partitions "${_pool}" size)
string(JSON _meta ERROR_VARIABLE _meta_error GET "${BUILD_PROFILE}" layout_info metadata_partition)
if(_meta_error)
    set(_meta "app_nvs")
endif()
math(EXPR _app_limit "${_slot_size} - ${_reserve}")
target_compile_definitions(${COMPONENT_LIB} PUBLIC AGENT_SYSTEM_ENABLED=1
    AGENT_SYSTEM_METADATA_PARTITION="${_meta}"
    AGENT_SYSTEM_BOARD_MODEL="${_model}" AGENT_SYSTEM_LAYOUT=${_layout}
    AGENT_SYSTEM_SLOT_COUNT=${_slots} AGENT_SYSTEM_COVER_RESERVE=${_reserve}
    AGENT_SYSTEM_APP_MAX_BYTES=${_app_limit})
target_compile_definitions(${COMPONENT_LIB} PRIVATE
    AGENT_SYSTEM_APP_ID="${IMAGE_app_id}" AGENT_SYSTEM_APP_NAME="${IMAGE_name}")
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-missing-field-initializers)
if(PROFILE_hardware STREQUAL "sdgoods-esp32s3")
    target_compile_definitions(${COMPONENT_LIB} PUBLIC AGENT_SYSTEM_SDGOODS_COMPAT=1)
endif()
